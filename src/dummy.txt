import axios, { AxiosRequestConfig, AxiosResponse } from "axios";
import * as microsoftTeams from "@microsoft/teams-js";
import { ensureTeamsJSInitialized } from "../../app/teams"; // Adjust this relative path to your index.ts

let _isProductionServer: boolean = false;

export const isProductionServer = (isProd: boolean) => {
  _isProductionServer = isProd;
};

export const isAuthRequired = () => {
  return _isProductionServer ? true : false;
};

// Returns both Primary (Azure) and Fallback (m365services) base URLs
export const getBaseURLs = () => {
  const primaryUrl = _isProductionServer
    ? "https://abc/"
    : "https://xyz/";

  const fallbackUrl = "https://xxx/";

  return { primaryUrl, fallbackUrl };
};

export class AxiosJWTDecorator {

  /**
   * Helper method to execute HTTP calls with Primary -> Fallback retry logic
   */
  private async executeWithFallback<R>(
    method: "get" | "post" | "put" | "delete",
    endpoint: string,
    data?: any,
    config?: AxiosRequestConfig
  ): Promise<R> {
    const isDirectCall = config?.data?.isDirectCall;

    // Direct call handling (no base URL attachment)
    if (isDirectCall) {
      return await axios[method](endpoint, data, config) as unknown as R;
    }

    const { primaryUrl, fallbackUrl } = getBaseURLs();
    const primaryFullUrl = primaryUrl + endpoint;
    const fallbackFullUrl = fallbackUrl + endpoint;

    // Step 1: Attempt Primary URL (PRD / DEV Azure)
    try {
      if (method === "get" || method === "delete") {
        return await axios[method](primaryFullUrl, config) as unknown as R;
      } else {
        return await axios[method](primaryFullUrl, data, config) as unknown as R;
      }
    } catch (primaryError) {
      console.warn(`Primary endpoint failed (${primaryFullUrl}). Retrying with fallback (${fallbackFullUrl})...`, primaryError);

      // Step 2: Fallback to m365services URL
      try {
        if (method === "get" || method === "delete") {
          return await axios[method](fallbackFullUrl, config) as unknown as R;
        } else {
          return await axios[method](fallbackFullUrl, data, config) as unknown as R;
        }
      } catch (fallbackError) {
        this.handleError(fallbackError);
        throw fallbackError;
      }
    }
  }

  // --- HTTP PUBLIC METHODS ---

  public async get<T = any, R = AxiosResponse<T>>(
    url: string,
    config?: AxiosRequestConfig,
    needAuthorizationHeaderTMP: boolean = true
  ): Promise<R> {
    config = await this.prepareConfig(config);
    return this.executeWithFallback<R>("get", url, undefined, config);
  }

  public async post<T = any, R = AxiosResponse<T>>(
    url: string,
    data?: any,
    config?: AxiosRequestConfig,
    needAuthorizationHeaderTMP: boolean = true
  ): Promise<R> {
    config = await this.prepareConfig(config);
    return this.executeWithFallback<R>("post", url, data, config);
  }

  public async put<T = any, R = AxiosResponse<T>>(
    url: string,
    data?: any,
    config?: AxiosRequestConfig,
    needAuthorizationHeaderTMP: boolean = true
  ): Promise<R> {
    config = await this.prepareConfig(config);
    return this.executeWithFallback<R>("put", url, data, config);
  }

  public async delete<T = any, R = AxiosResponse<T>>(
    url: string,
    data?: any,
    config?: AxiosRequestConfig
  ): Promise<R> {
    config = await this.prepareConfig(config);

    if (data) {
      config = config || {};
      config.headers = config.headers || {};
      config.headers["Content-Type"] = "application/json; charset=utf-8";
      config.data = data;
    }

    return this.executeWithFallback<R>("delete", url, undefined, config);
  }

  // --- PRIVATE HELPERS ---

  private async prepareConfig(config?: AxiosRequestConfig): Promise<AxiosRequestConfig> {
    let conf = config || {};
    const needAuth = isAuthRequired() || conf?.data?.isAPIAuthRequired;

    if (needAuth) {
      conf = await this.setupAuthorizationHeader(conf);
    }
    return conf;
  }

  // UPDATED FOR TEAMS SDK 2.x & MOBILE COMPATIBILITY
  private async setupAuthorizationHeader(config: AxiosRequestConfig): Promise<AxiosRequestConfig> {
    try {
      // Safely await Teams SDK init without re-initializing manually
      await ensureTeamsJSInitialized();

      if (microsoftTeams.authentication.isSupported()) {
        // Teams JS 2.x Promise-based getAuthToken
        const token = await microsoftTeams.authentication.getAuthToken({
          resources: ["https://graph.microsoft.com"]
        });

        config = config || {};
        config.headers = config.headers || {};
        config.headers["Authorization"] = `Bearer ${token}`;
      }
    } catch (error) {
      console.error("Error from getAuthToken:", error);
      window.location.href = "/signin";
    }

    return config;
  }

  private handleError(error: any): void {
    console.error("Axios-Error", error);

    if (error?.hasOwnProperty("response") || error?.response) {
      const errorStatus = error.response?.status;
      if (errorStatus === 403 || errorStatus === 401) {
        window.location.href = "/errorpage";
      } else {
        window.location.href = "/errorpage";
      }
    } else {
      window.location.href = "/errorpage";
    }
  }
}

export const axiosJWTDecoratorInstance = new AxiosJWTDecorator();
export default axiosJWTDecoratorInstance;
