import React, { useState, useEffect, useMemo, useCallback } from 'react';
import { useForm, Controller } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { z } from 'zod';

// ==========================================
// 1. Zod Validation Schemas & Types
// ==========================================

export const SPOTeamsExternalSharingGetDataSchema = z.object({
  webAddDropdown: z
    .string({ required_error: 'Please select value from the dropdown' })
    .min(1, 'Website address dropdown is required'),
  siteUrl: z
    .string({ required_error: 'Please provide site URL' })
    .min(1, 'Please provide site URL')
});

export type SPOTeamsExternalSharingGetDataValues = z.infer<
  typeof SPOTeamsExternalSharingGetDataSchema
>;

export const SPOTeamsExternalSharingSetDataSchema = z.object({
  State: z
    .string({ required_error: 'Please choose at least one state' })
    .min(1, 'Please choose at least one state'),
  AllowDomains: z
    .string({ required_error: 'Please provide domain name(s)' })
    .min(1, 'Please provide domain name(s)')
    .refine((domains) => validateDomains(domains), {
      message: 'One or more domains are invalid or improperly separated.'
    }),
  Reason: z
    .string({ required_error: 'Please provide reason' })
    .min(1, 'Please provide reason')
});

export type SPOTeamsExternalSharingSetDataValues = z.infer<
  typeof SPOTeamsExternalSharingSetDataSchema
>;

// Helper to convert backend status code to Radio option value
const mapStatusToRadio = (status?: string): 'Disable' | 'ExternalUserSharingOnly' => {
  return status === '0' ? 'Disable' : 'ExternalUserSharingOnly';
};

// ==========================================
// 2. Main Unified Component
// ==========================================

export const SPOUserExternalSharingModule: React.FC = () => {
  const { screenWidth } = useScreen();

  const screenOrientation = useMemo<'horizontal' | 'vertical'>(
    () => (screenWidth > ScreenSize.tablet ? 'horizontal' : 'vertical'),
    [screenWidth]
  );

  // Global Notification State
  const [msg, setMsg] = useState<StatusMessageProps>({
    message: '',
    status: Status.SUCCESS,
    type: 'success'
  });

  // Track verified backend data & service execution state
  const [boundData, setBoundData] = useState<{
    allowedDomain: string;
    status: string;
    verifiedUrl: string;
    isServiceCalled: boolean;
  }>({
    allowedDomain: '',
    status: '0',
    verifiedUrl: '',
    isServiceCalled: false
  });

  // --- Service 1: Get Data / Verification Mutation ---
  const [
    fetchSiteData,
    {
      data: getDataResponse,
      isLoading: isGetDataLoading,
      isSuccess: isGetDataSuccess,
      isError: isGetDataError
    }
  ] = useSPOTeamsExSharingGetDataMutation();

  // --- Service 2: Set Data / Submission Mutation ---
  const [
    submitSiteData,
    {
      isLoading: isSetDataLoading,
      isSuccess: isSetDataSuccess,
      isError: isSetDataError
    }
  ] = useSPOTeamsExSharingSetDataMutation();

  // --- Form 1: Site URL Form ---
  const urlForm = useForm<SPOTeamsExternalSharingGetDataValues>({
    resolver: zodResolver(SPOTeamsExternalSharingGetDataSchema),
    defaultValues: {
      webAddDropdown: 'https://share.novartis.net/sites/',
      siteUrl: ''
    },
    mode: 'onChange'
  });

  // --- Form 2: Operations Bottom Form ---
  const operationForm = useForm<SPOTeamsExternalSharingSetDataValues>({
    resolver: zodResolver(SPOTeamsExternalSharingSetDataSchema),
    defaultValues: {
      AllowDomains: '',
      Reason: '',
      State: 'Disable'
    },
    mode: 'onChange'
  });

  // ==========================================
  // Handlers & Side Effects
  // ==========================================

  // Sync API response to bottom form & verification state
  useEffect(() => {
    if (isGetDataSuccess && getDataResponse) {
      const fullUrl =
        urlForm.getValues('webAddDropdown') + urlForm.getValues('siteUrl');

      setBoundData({
        allowedDomain: getDataResponse.allowedDomain || '',
        status: getDataResponse.status || '0',
        verifiedUrl: fullUrl,
        isServiceCalled: true
      });

      // Clear any previous error states and populate received values
      operationForm.clearErrors();
      urlForm.clearErrors();
      operationForm.reset({
        AllowDomains: getDataResponse.allowedDomain || '',
        State: mapStatusToRadio(getDataResponse.status),
        Reason: ''
      });

      setMsg({
        status: Status.SUCCESS,
        type: 'success',
        message: 'Site verified successfully.'
      });
    } else if (isGetDataError) {
      setMsg({
        status: Status.ERROR,
        type: 'error',
        message: 'Failed to fetch site information.'
      });
    }
  }, [isGetDataSuccess, isGetDataError, getDataResponse, urlForm, operationForm]);

  // Handle Submit (Service 2) Notifications
  useEffect(() => {
    if (!isSetDataLoading && isSetDataSuccess) {
      setMsg({
        status: Status.SUCCESS,
        type: 'success',
        message: 'Operation completed successfully.'
      });
      operationForm.resetField('Reason');
    } else if (!isSetDataLoading && isSetDataError) {
      setMsg({
        status: Status.ERROR,
        type: 'error',
        message: 'Error while updating records.'
      });
    }
  }, [isSetDataLoading, isSetDataSuccess, isSetDataError, operationForm]);

  // Action 1: "Get Data" Button Click
  const handleGetDataSubmit = useCallback(
    async (e: React.MouseEvent) => {
      e.preventDefault();
      // Ensure bottom errors are cleared when running URL validation
      operationForm.clearErrors();

      const isValid = await urlForm.trigger();
      if (isValid) {
        const { webAddDropdown, siteUrl } = urlForm.getValues();
        fetchSiteData({
          siteCollectionURL: webAddDropdown + siteUrl
        });
      }
    },
    [urlForm, operationForm, fetchSiteData]
  );

  // Action 2: "Submit" Button Click
  const handleSetDataSubmit = useCallback(
    async (e: React.MouseEvent) => {
      e.preventDefault();

      // CASE 1: "Get Data" has NOT been called yet
      if (!boundData.isServiceCalled || !boundData.verifiedUrl) {
        // Clear bottom form errors so only dropdown & siteUrl highlight
        operationForm.clearErrors();
        await urlForm.trigger();

        setMsg({
          status: Status.ERROR,
          type: 'error',
          message: 'Please validate the site URL first by clicking "Get Data".'
        });
        return;
      }

      // Check if user edited the URL after verifying
      const currentUrl =
        urlForm.getValues('webAddDropdown') + urlForm.getValues('siteUrl');
      if (currentUrl !== boundData.verifiedUrl) {
        operationForm.clearErrors();
        await urlForm.trigger();

        setMsg({
          status: Status.ERROR,
          type: 'error',
          message: 'URL was modified. Please click "Get Data" again to revalidate.'
        });
        return;
      }

      // CASE 2: "Get Data" WAS called and verified
      // Clear URL form errors and validate ONLY bottom fields
      urlForm.clearErrors();
      const isValid = await operationForm.trigger();
      if (isValid) {
        const formValues = operationForm.getValues();
        submitSiteData({
          siteCollectionURL: boundData.verifiedUrl,
          ...formValues
        });
      }
    },
    [boundData, urlForm, operationForm, submitSiteData]
  );

  // Reset Bottom Section back to initial verified data
  const handleResetBottomSection = useCallback(() => {
    operationForm.reset({
      AllowDomains: boundData.allowedDomain,
      State: mapStatusToRadio(boundData.status),
      Reason: ''
    });
    operationForm.clearErrors();
  }, [boundData, operationForm]);

  // Invalidate verification when the URL or dropdown is altered
  const handleUrlInputChange = (val: string, fieldOnChange: (v: string) => void) => {
    fieldOnChange(val);
    if (boundData.isServiceCalled) {
      setBoundData((prev) => ({
        ...prev,
        isServiceCalled: false,
        verifiedUrl: ''
      }));
    }
  };

  // ==========================================
  // Render
  // ==========================================

  return (
    <div className="guest-user-deletion-base space-y-6">
      {/* Toast Notification */}
      {msg.message && <StatusMessageComponent {...msg} />}

      {/* Top Section: URL Form */}
      <div className="guest-user-deletion-container">
        <Field
          className="pb-6"
          orientation={screenOrientation}
          label="Website Address"
          validationMessage={
            urlForm.formState.errors?.webAddDropdown?.message ||
            urlForm.formState.errors?.siteUrl?.message
          }
          validationState={
            urlForm.formState.errors?.webAddDropdown ||
            urlForm.formState.errors?.siteUrl
              ? 'error'
              : 'none'
          }
        >
          <div className="flex w-full gap-x-2">
            <Controller
              name="webAddDropdown"
              control={urlForm.control}
              render={({ field: { onChange, value } }) => (
                <div className="flex">
                  <CommonDropdown
                    defaultSelectedOptions={['sn']}
                    placeholder="Select site"
                    onOptionSelect={(_, d) => {
                      handleUrlInputChange(d.optionValue, onChange);
                    }}
                    value={value}
                  >
                    <Option key={1} value="https://share.novartis.net/sites/">
                      https://share.novartis.net/sites/
                    </Option>
                  </CommonDropdown>
                </div>
              )}
            />

            <Controller
              name="siteUrl"
              control={urlForm.control}
              render={({ field: { onChange, value } }) => (
                <div className="flex w-full">
                  <Input
                    className="block w-full"
                    placeholder="Site URL"
                    value={value}
                    onChange={(_, d) => handleUrlInputChange(d.value, onChange)}
                  />
                </div>
              )}
            />
          </div>
        </Field>

        <div className="w-full flex justify-end">
          <CommonButton
            disabled={isGetDataLoading}
            onClick={handleGetDataSubmit}
          >
            {isGetDataLoading ? 'Validating...' : 'Get Data'}
          </CommonButton>
        </div>
      </div>

      {/* Bottom Section: Operations Form */}
      <div className="guest-user-deletion-container">
        {/* State / Operation Radio Group */}
        <Field
          className="pb-6"
          orientation={screenOrientation}
          label="Operation"
          validationMessage={operationForm.formState.errors?.State?.message}
          validationState={operationForm.formState.errors?.State ? 'error' : 'none'}
        >
          <Controller
            name="State"
            control={operationForm.control}
            render={({ field: { onChange, value } }) => (
              <RadioGroup
                layout="vertical"
                value={value}
                onChange={(_, d) => onChange(d.value)}
              >
                <Radio value="Disable" label="Disable" />
                <Radio
                  value="ExternalUserSharingOnly"
                  label="Enable (External User Sharing Only)"
                />
              </RadioGroup>
            )}
          />
        </Field>

        {/* Allowed Domains Textarea */}
        <Field
          className="pb-6"
          orientation={screenOrientation}
          label="Allow only specific domain"
          validationMessage={operationForm.formState.errors?.AllowDomains?.message}
          validationState={
            operationForm.formState.errors?.AllowDomains ? 'error' : 'none'
          }
        >
          <Controller
            name="AllowDomains"
            control={operationForm.control}
            render={({ field: { onChange, value } }) => (
              <Textarea
                cols={3}
                className="block w-full"
                value={value}
                onChange={(_, d) => onChange(d.value)}
              />
            )}
          />
        </Field>

        {/* Reason Textarea */}
        <Field
          className="pb-6"
          orientation={screenOrientation}
          label="Reason"
          validationMessage={operationForm.formState.errors?.Reason?.message}
          validationState={
            operationForm.formState.errors?.Reason ? 'error' : 'none'
          }
        >
          <Controller
            name="Reason"
            control={operationForm.control}
            render={({ field: { onChange, value } }) => (
              <Textarea
                cols={3}
                className="block w-full"
                resize="both"
                value={value}
                onChange={(_, d) => onChange(d.value)}
              />
            )}
          />
        </Field>

        {/* Action Buttons */}
        <div className="flex justify-end gap-3">
          <CommonButton
            disabled={isSetDataLoading}
            onClick={handleResetBottomSection}
          >
            Reset
          </CommonButton>
          <CommonButton
            appearance="primary"
            disabled={isSetDataLoading}
            onClick={handleSetDataSubmit}
          >
            {isSetDataLoading ? 'Submitting...' : 'Submit'}
          </CommonButton>
        </div>
      </div>
    </div>
  );
};

export default SPOUserExternalSharingModule;
