import { Outlet } from 'react-router-dom';

export const ResizableLayoutModule: React.FC = () => {
  const { isValidModule } = useGetCurrentModuleName();
  const [isCollapse, setIsCollapse] = useState<boolean>(true);
  const [navWidth, setNavWidth] = useState<number>(250);
  const [isResizing, setIsResizing] = useState<boolean>(false);
  const { isMobile } = useScreen();

  // Refs prevent re-binding mousemove listeners on every pixel dragged
  const dragStartXRef = useRef<number>(0);
  const dragStartWidthRef = useRef<number>(250);

  // Query hooks
  const { isLoading: isMenuLoading } = useGetMenusQuery(null);
  const { isLoading: isUserInfoLoading } = useGetUserInfoQuery(null);
  const { isLoading: isRoleLoading } = useGetUserRoleQuery(null);

  const isBootstrapping = isUserInfoLoading || isRoleLoading || isMenuLoading;

  useEffect(() => {
    if (!isMobile && !isCollapse) {
      setNavWidth(250);
    } else if (isCollapse || isMobile) {
      setNavWidth(60);
    }
  }, [isCollapse, isMobile]);

  const handleMouseDown = useCallback(
    (e: React.MouseEvent<HTMLDivElement>) => {
      if (isCollapse || isMobile) return;
      e.preventDefault();
      dragStartXRef.current = e.pageX;
      dragStartWidthRef.current = navWidth;
      setIsResizing(true);
    },
    [isCollapse, isMobile, navWidth]
  );

  useEffect(() => {
    if (!isResizing) return;

    const handleMouseMove = (e: MouseEvent) => {
      const diff = e.pageX - dragStartXRef.current;
      const newWidth = Math.max(200, Math.min(500, dragStartWidthRef.current + diff));
      setNavWidth(newWidth);
    };

    const handleMouseUp = () => {
      setIsResizing(false);
    };

    document.body.classList.add('select-none', 'cursor-col-resize');
    document.addEventListener('mousemove', handleMouseMove);
    document.addEventListener('mouseup', handleMouseUp);

    return () => {
      document.body.classList.remove('select-none', 'cursor-col-resize');
      document.removeEventListener('mousemove', handleMouseMove);
      document.removeEventListener('mouseup', handleMouseUp);
    };
  }, [isResizing]);

  // Suppress benign ResizeObserver overlay errors with proper cleanup
  useEffect(() => {
    const handleResizeObserverError = (e: ErrorEvent) => {
      if (
        e.message === 'ResizeObserver loop limit exceeded' ||
        e.message === 'ResizeObserver loop completed with undelivered notifications.'
      ) {
        const overlayDiv = document.getElementById('webpack-dev-server-client-overlay-div');
        const overlay = document.getElementById('webpack-dev-server-client-overlay');
        if (overlayDiv) overlayDiv.style.display = 'none';
        if (overlay) overlay.style.display = 'none';
      }
    };

    window.addEventListener('error', handleResizeObserverError);
    return () => window.removeEventListener('error', handleResizeObserverError);
  }, []);

  const preloadMessage = isRoleLoading
    ? 'Loading User Roles...'
    : isUserInfoLoading
    ? 'Loading User Information...'
    : isMenuLoading
    ? 'Loading Menubar...'
    : '';

  if (!isValidModule) {
    return <NotFoundLayout />;
  }

  if (isBootstrapping) {
    return <LayoutSkeleton message={preloadMessage} />;
  }

  return (
    <div className="flex flex-col h-screen">
      {/* Fixed Header */}
      <header className="resizable-layout-header bg-(--colorNeutralBackground1) flex-none h-[55px] max-[635px]:h-[80px] border-b border-gray-200 fixed top-0 inset-x-0 z-50 shadow-md">
        <HeaderComponent />
      </header>

      {/* Main Content Area */}
      <div className="flex flex-1 min-h-0 pt-[55px] max-[635px]:pt-[80px]">
        {/* Navbar with Resize Handle */}
        <aside
          className="resizable-layout-navbar bg-(--colorNeutralBackground1) flex-none relative transition-[width] duration-150 ease-out"
          style={{
            width: `${navWidth}px`,
            transitionProperty: isResizing ? 'none' : 'width',
          }}
        >
          <NavbarComponent
            className="w-full h-full"
            onCollapse={setIsCollapse}
          />

          {/* Resize Handle (Hidden when collapsed or on mobile) */}
          {!isCollapse && !isMobile && (
            <div
              role="separator"
              aria-orientation="vertical"
              aria-label="Resize sidebar"
              onMouseDown={handleMouseDown}
              className="absolute top-0 right-0 w-1 h-full cursor-col-resize bg-gray-300 hover:bg-blue-500 active:bg-blue-600 transition-colors"
            />
          )}
        </aside>

        {/* Main Content Container */}
        <main className="resizable-layout-container bg-(--colorNeutralBackground1) flex-1 overflow-auto p-1">
          <UIReloadNotificationBar />
          <Outlet />
        </main>
      </div>

      {/* Footer */}
      <footer className="resizable-layout-footer bg-(--colorNeutralBackground1) flex-none h-16 border-t text-white flex items-center justify-center shadow-inner">
        {/* Footer Content */}
      </footer>

      <Toastr />
    </div>
  );
};

export default ResizableLayoutModule;
