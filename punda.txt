import React from 'react';
import { useOutlet } from 'react-router-dom';
import { M365ToolsCollapsiblePanels } from './components/m365Tools-collapsible-panels.component';

  const outlet = useOutlet();

  return (
    <div className="m365-tools-container">

        heading={
          <div className="flex items-center justify-start pb-3">

              M365 Administration Tools
            </h4>
          </div>
        }
        showSearch={false}
      />
      <hr />
      {/* Renders child route if matched; falls back to collapsible panels if not */}
      {outlet || <M365ToolsCollapsiblePanels />}
    </div>
  );
};

export default M365ManagementTools;
